4.5不确定性收敛的判定逻辑（程序实现）
代码块
def check_uncertainty_convergence(uncertainty, hypothesis_tree, 
experiment_results): 
    """ 
    判断⼀个不确定性是否已收敛（解决） 
    """ 
    # 条件 1 ：相关假设已被剪枝 
    for h in uncertainty.related_hypotheses: 
        if hypothesis_tree.get(h).status == "pruned": 
            uncertainty.status = "deprecated" 
            uncertainty.resolution = f" 假设{h}已被剪枝，不确定性⾃动失效 " 
            return True 
     
    # 条件 2 ：相关假设已收敛且差距明确 
    supports = [hypothesis_tree.get(h).support_score for h in 
uncertainty.related_hypotheses] 
    if all(s >= 0.70 for s in supports): 
        uncertainty.status = "resolved" 
        uncertainty.resolution = " 所有相关假设均已收敛 " 
        return True 
     
    # 条件 3 ：存在明确的区分实验结果 
    for exp in experiment_results: 
        if exp.design.resolves_uncertainty == uncertainty.uncertainty_id: 
            if exp.result.is_conclusive(): 
                uncertainty.status = "resolved" 
                uncertainty.resolution = exp.result.conclusion 
                uncertainty.resolving_experiment = exp.experiment_id 
                return True 
     
    return False
