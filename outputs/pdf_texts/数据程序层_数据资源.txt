class DataManager: 
    """ 统⼀数据访问管理器 """ 
     
    def __init__(self, data_dir: str): 
        self.data_dir = data_dir 
        self.dictionary = self._load_dictionary() 
     
    def get_dictionary(self) -> dict: 
        """ 返回完整数据字典 """ 
        return self.dictionary 
     
    def get_available_fields(self) -> list: 
        """ 返回所有可⽤字段列表 """ 
        return [f['field_name'] for f in self.dictionary['fields']] 
     
    def get_target_fields(self) -> list: 
        """ 返回候选⽬标变量列表 """ 
        return self.dictionary.get('target_candidates', []) 
     
    def get_feature_fields(self) -> list: 
        """ 返回候选特征变量列表 """

    def get_field_metadata(self, field_name: str) -> dict: 
        """ 返回特定字段的元数据 """ 
        for f in self.dictionary['fields']: 
            if f['field_name'] == field_name: 
                return f 
        return None 
     
    def load_data(self, columns: list = None) -> pd.DataFrame: 
        """ 加载处理后的数据，可选择指定列 """ 
        df = pd.read_parquet(self.data_dir / 'processed/aligned_data.parquet') 
        if columns: 
            return df[columns] 
        return df 
     
    def get_sample_count(self) -> int: 
        return self.dictionary['total_samples']

5.2数据字典的版本管理
数据字典需要版本控制：
代码块
data/dictionary/ 
├──  data_dictionary.json           # 软链接，指向最新版本 
├──  data_dictionary_v1.0.json 
├──  data_dictionary_v1.1.json 
└──  CHANGELOG.md                   # 记录变更历史 
版本号规则：v{major}.{minor} ，主要变更（如新增⽂件、更换数据集）时升major，局部调整
（如补充字段说明）时升minor。
版本升级规则：数据字典版本号仅由数据上传事件触发，每次⽤⼾上传⽂件或主动更新字典时版本递
增，与实验轮次⽆关。
