def prediction_to_distribution(lower, upper, method='normal'): 
    """ 
    将预测区间转换为概率分布 
    """ 
    mean = (lower + upper) / 2 
    std = (upper - lower) / 4  # 假设 95% 置信区间，即  ±2 σ 
     
    if method == 'normal': 
        return scipy.stats.norm(mean, std) 
    else:  # uniform 
        return scipy.stats.uniform(lower, upper - lower)

在没有真实实验结果时，⽤各假设预测分布的重叠程度来近似信息增益：重叠越少→实验越能区分假

3.2计算公式
对于任意两个假设H_i和H_j：
Step1：计算分布重叠⾯积（KL散度的简化替代）
代码块
overlap(H_i, H_j) = ∫  min(P_i(x), P_j(x)) dx 
其中P_i(x)和P_j(x)是两假设预测结果的概率密度函数。
Step2：归⼀化重叠度
代码块
normalized_overlap(H_i, H_j) = overlap(H_i, H_j) / max( ∫ P_i(x)dx, ∫ P_j(x)dx) 
注：归⼀化后的取值范围为[0,1]。
Step3：计算假设对之间的信息增益
代码块
IG_pair(H_i, H_j) = 1 - normalized_overlap(H_i, H_j) 
Step4：多假设场景（n>2）
代码块
IG(E) = (2 / (n*(n-1))) * Σ _{i<j} IG_pair(H_i, H_j) 
即所有假设对信息增益的平均值。

def calculate_overlap_area(p1_mean, p1_std, p2_mean, p2_std): 
    """ 
    计算两个正态分布的重叠⾯积（ KL 散度近似） 
    """ 
    # 两分布的交叉点 
    a = (p1_mean**2 - p2_mean**2) / (2 * (p1_mean - p2_mean)) 
     
    # 计算每个分布的另⼀半⾯积 
    def min_pdf(x): 
        return min(norm.pdf(x, p1_mean, p1_std),  
                   norm.pdf(x, p2_mean, p2_std)) 
     
    # 数值积分求重叠⾯积 
    overlap, _ = quad(min_pdf, -np.inf, np.inf) 
    return overlap 
 
def calculate_information_gain_round1(experiment, hypotheses): 
    """ 
    第⼀轮信息增益计算：重叠⾯积近似法 
    """ 
    n = len(hypotheses) 
    if n < 2: 
        return 0.5  # 只有 1 个假设，信息增益⽆法定义，取 0.5 
     
    total_ig = 0 
    pair_count = 0 
     
    for i in range(n): 
        for j in range(i+1, n): 
            # 获取两假设的预测分布参数 
            H_i = hypotheses[i] 
            H_j = hypotheses[j] 
             
            # 将预测区间转换为分布 
            mu_i = (H_i.lower + H_i.upper) / 2 
            sigma_i = (H_i.upper - H_i.lower) / 4 
             
            mu_j = (H_j.lower + H_j.upper) / 2 
            sigma_j = (H_j.upper - H_j.lower) / 4 
             
            # 计算重叠⾯积 
            overlap = calculate_overlap_area(mu_i, sigma_i, mu_j, sigma_j) 
             
            # 归⼀化 
            max_area = max(norm.pdf(mu_i, mu_i, sigma_i),  
                          norm.pdf(mu_j, mu_j, sigma_j)) 
            normalized_overlap = overlap / max_area

Step2：计算似然（实际结果在假设预测分布下的概率）
代码块
L(Y_actual | H_i) = P(Y_actual | H_i) ≈  PDF_i(Y_actual) 
其中PDF_i是假设H_i的预测概率密度函数（正态分布）。
Step3：计算后验概率（⻉叶斯更新）
代码块
P(H_i | Y_actual) = L(Y_actual | H_i) * P(H_i) / Σ _j L(Y_actual | H_j) * P(H_j) 
Step4：计算KL散度（信息增益）
代码块
IG(E) = KL(P_posterior || P_prior) = Σ _i P(H_i|Y_actual) * log(P(H_i|Y_actual) 
/ P(H_i))

import numpy as np 
from scipy.stats import norm 
 
def calculate_information_gain_round_n(experiment, actual_result, hypotheses, 
prior_supports): 
    """ 
    第 N 轮（ N ≥ 2 ）信息增益计算： KL 散度法 
    """ 
    # Step 1: 计算先验概率 
    total_prior = sum(prior_supports) 
    prior_probs = [s / total_prior for s in prior_supports] 
     
    # Step 2: 计算似然 
    likelihoods = [] 
    for h in hypotheses: 
        # 将预测区间转换为分布 
        mu = (h.lower + h.upper) / 2 
        sigma = (h.upper - h.lower) / 4 
        # 计算实际结果在该假设下的概率密度 
        lik = norm.pdf(actual_result, mu, sigma)

prior_probs)] 
    total_posterior = sum(unnormalized_posterior)

        if post > 0 and prior > 0: 
            ig += post * np.log(post / prior) 
     
    return ig, posterior_probs

def calculate_llm_cost(experiment, hypothesis_tree, budget): 
    """ 
    计算 LLM Token 成本 
    """ 
    # 1. 估算输⼊ Token 
    input_tokens = 2000  # 基础系统提⽰词 
     
    # 假设树状态 
    active_hypotheses = hypothesis_tree.get_active() 
    for h in active_hypotheses:

    steps = len(experiment.get('steps', [])) 
    step_coeff = 1 + (steps - 1) * 0.2 
     
    # 4. 计算数据规模系数 
    data_coeff = max(1, (n_samples / 1000) * (max(n_features, 1) / 20)) 
     
    # 5. 基准时间 
    base_time = 10  # 秒 
     
    estimated_seconds = base_time * data_coeff * model_coeff * step_coeff 
     
    # 6. 增加 XGBoost 的额外惩罚（调参开销） 
    if model == 'XGBoost' and experiment['design'].get('tuning', False): 
        estimated_seconds *= 1.5

min( 估算 Token 数  / Token 预算上限 , 1.0) 。对于C2（计算成本），归⼀化⽅式为 min( 估算

预算上限的值定义在 system_config.yaml 中。为避免单轮成本评估出现极端值，程序会对归⼀
化后的值做平滑处理（例如设置最⼩值0.05或0.1，代表“即使最简单的实验也有基础成本”），并
在综合价值计算中直接使⽤归⼀化后的Cost(E)。

def calculate_utility_cost(experiment, budget): 
    """ 
    计算最终的归⼀化成本 
    """ 
    # 获取预算上限 
    token_budget = budget.get('max_tokens_per_round', 200000)

    # 2. C2: 计算成本归⼀化 
    c2_raw = calculate_compute_cost(experiment) 
    c2_norm = min(c2_raw / time_budget, 1.0)

（5）性能增益评估程序
性能增益回答的问题是：“做这个实验，预测精度能提升多少？”
⼆、核⼼定义与计算公式
2.1核⼼定义
性能增益PG(E)=实验E预期能带来的预测性能提升幅度，归⼀化到[0,1]区间。
其中：
•
RMSE_baseline ：当前最佳性能（已实现的最佳RMSE）
•
RMSE_predicted(E) ：实验E预期的RMSE
•
RMSE_ideal ：理论上限（不可达到，设为0或历史最低值）
2.2计算流程（两阶段）
实验前⽤于综合价值评估（选择实验），实验后⽤于审计和学习（更新预测模型）。
三、第⼀轮的特殊处理
3.1为什么第⼀轮⽆法计算PG

U(E) = α ·IG(E) − γ ·Risk(E) − δ ·Cost(E)

从第⼆轮开始，PG可作为综合价值的组成部分。β 的值定义在 system_config.yaml 中。

    risk = calculate_risk(experiment, state)
